export const getTeamDetails = async (req, res) => {
  const { id } = req.params;
  try {
    const team = await teamService.getTeamById(id);
    if (!team) {
      return res.status(404).json({ message: "Equipo no encontrado." });
    }
    res.status(200).json(team);
